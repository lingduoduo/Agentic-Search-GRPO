# rather than failing at startup when they are absent.
#
#   pip install -r requirements.txt -r requirements-retrieval-heavy.txt
#
# faiss-cpu on macOS: install via conda to avoid an OpenMP conflict with PyTorch
#   conda install -c conda-forge faiss-cpu
# pip install faiss-cpu works on Linux but segfaults on macOS Apple Silicon.
faiss-cpu==1.13.2
